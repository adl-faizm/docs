XLSmart Design System is published as an npm package and supports all major React frameworks. This page covers framework-specific setup steps beyond the basic installation covered in the [Quickstart](/quickstart).

## Requirements

- **React** 17.0.0 or higher
- **Node.js** 16.0.0 or higher
- **TypeScript** 4.5+ (optional, but recommended — the package ships with full types)

## Next.js

<Steps>
  <Step title="Install the package">
    ```bash npm
    npm install @xlsmart/design-system
    ```
  </Step>

  <Step title="Add global styles to layout">
    In `app/layout.tsx` (App Router) or `pages/_app.tsx` (Pages Router):

    <Tabs>
      <Tab title="App Router">
        ```tsx app/layout.tsx
        import '@xlsmart/design-system/styles/base.css';
        import type { ReactNode } from 'react';

        export default function RootLayout({ children }: { children: ReactNode }) {
          return (
            <html lang="en">
              <body>{children}</body>
            </html>
          );
        }
        ```
      </Tab>
      <Tab title="Pages Router">
        ```tsx pages/_app.tsx
        import '@xlsmart/design-system/styles/base.css';
        import type { AppProps } from 'next/app';

        export default function App({ Component, pageProps }: AppProps) {
          return <Component {...pageProps} />;
        }
        ```
      </Tab>
    </Tabs>
  </Step>

  <Step title="Use components">
    ```tsx app/page.tsx
    import { Button, Stack } from '@xlsmart/design-system';

    export default function Page() {
      return (
        <Stack gap={4}>
          <h1>Welcome</h1>
          <Button variant="primary">Get Started</Button>
        </Stack>
      );
    }
    ```
  </Step>
</Steps>

<Info>XLSmart Design System components are fully compatible with Next.js Server Components. Components that use React hooks (`useState`, `useEffect`) are automatically marked with `"use client"` internally — you do not need to add it yourself.</Info>

## Vite + React

<Steps>
  <Step title="Install the package">
    ```bash npm
    npm install @xlsmart/design-system
    ```
  </Step>

  <Step title="Import global styles">
    In `src/main.tsx`:

    ```tsx src/main.tsx
    import React from 'react';
    import ReactDOM from 'react-dom/client';
    import '@xlsmart/design-system/styles/base.css';
    import App from './App';

    ReactDOM.createRoot(document.getElementById('root')!).render(
      <React.StrictMode>
        <App />
      </React.StrictMode>
    );
    ```
  </Step>

  <Step title="Use components">
    ```tsx src/App.tsx
    import { Button } from '@xlsmart/design-system';

    export default function App() {
      return <Button variant="primary">Hello XLSmart</Button>;
    }
    ```
  </Step>
</Steps>

## Create React App

<Steps>
  <Step title="Install the package">
    ```bash npm
    npm install @xlsmart/design-system
    ```
  </Step>

  <Step title="Import global styles">
    In `src/index.tsx`:

    ```tsx src/index.tsx
    import React from 'react';
    import ReactDOM from 'react-dom/client';
    import '@xlsmart/design-system/styles/base.css';
    import App from './App';

    const root = ReactDOM.createRoot(document.getElementById('root') as HTMLElement);
    root.render(<App />);
    ```
  </Step>
</Steps>

## TypeScript support

The package ships with full TypeScript definitions. No additional `@types` packages are needed. Your IDE will provide autocomplete for all component props and token values.

```tsx typescript-example.tsx
import { Button } from '@xlsmart/design-system';
// ButtonProps is available for extending or typing references
import type { ButtonProps } from '@xlsmart/design-system';

const MyButton = (props: ButtonProps) => <Button {...props} />;
```
